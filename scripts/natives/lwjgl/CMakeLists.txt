cmake_minimum_required(VERSION 3.22)
project(cofre_lwjgl LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(modules "${LWJGL_SOURCE}/modules/lwjgl")
set(core "${modules}/core/src/main/c")
include_directories("${JAVA_HEADERS}" "${JAVA_HEADERS}/linux" "${core}" "${core}/linux")
add_compile_definitions(LWJGL_LINUX LWJGL_arm64 _GNU_SOURCE CONFIG_HAVE_MEMFD_CREATE)
file(GLOB core_sources "${core}/*.c" "${modules}/core/src/generated/c/*.c"
    "${modules}/core/src/generated/c/linux/*.c" "${core}/linux/liburing/*.c")
add_library(lwjgl SHARED ${core_sources})
target_include_directories(lwjgl PRIVATE "${FFI_INCLUDE}" "${core}/linux/liburing" "${core}/linux/liburing/include")
target_link_libraries(lwjgl PRIVATE "${FFI_LIBRARY}" dl m)
file(GLOB stb_sources "${modules}/stb/src/generated/c/*.c")
add_library(lwjgl_stb SHARED ${stb_sources})
target_include_directories(lwjgl_stb PRIVATE "${modules}/stb/src/main/c")
target_link_libraries(lwjgl_stb PRIVATE m)
file(GLOB vma_sources "${modules}/vma/src/generated/c/*.cpp")
add_library(lwjgl_vma SHARED ${vma_sources})
target_include_directories(lwjgl_vma PRIVATE "${modules}/vma/src/main/c" "${modules}/vulkan/src/main/c")
file(GLOB tinyfd_sources "${modules}/tinyfd/src/generated/c/*.c" "${modules}/tinyfd/src/main/c/*.c")
add_library(lwjgl_tinyfd SHARED ${tinyfd_sources})
target_include_directories(lwjgl_tinyfd PRIVATE "${modules}/tinyfd/src/main/c")
# MC 26.2's bootstrap loads GL even for Vulkan. This is JNI only, no GL driver.
file(GLOB gl_sources "${modules}/opengl/src/generated/c/*.c")
list(FILTER gl_sources EXCLUDE REGEX "_WGL\\.c$")
add_library(lwjgl_opengl SHARED ${gl_sources})
target_include_directories(lwjgl_opengl PRIVATE "${modules}/opengl/src/main/c")
foreach(target lwjgl lwjgl_stb lwjgl_vma lwjgl_tinyfd lwjgl_opengl)
    target_link_options(${target} PRIVATE -Wl,--no-undefined "-Wl,--version-script=${LWJGL_SOURCE}/config/linux/version.script")
endforeach()
